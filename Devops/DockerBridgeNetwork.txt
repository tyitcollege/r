Eg1.java

public class Eg1{
    public static void main(String args[]){
        int a = 5;

        if (a % 2 == 0){
            System.out.println("Even");
        }
        else{
            System.out.println("Odd");
        }
    }
}

FROM eclipse-temurin:17-jdk

WORKDIR /devoops/app

COPY Eg1.java ./

RUN javac Eg1.java

CMD ["java", "Eg1"]
