area_of_circle.py

import sys
import math
import time

radius = 5.0

if len(sys.argv) > 1:
    radius = float(sys.argv[1])

area = math.pi * radius ** 2
print(f"The area of a circle with radius {radius} is: {area}", flush=True)

time.sleep(3600)
