eg1.java

public class eg1 {
    public static void main(String args[]) {
        int a = 7;

        if(a % 2 == 0) {
            System.out.println(a + " is an even number");
        } 
        else {
            System.out.println(a + " is an odd number");
        }

        try {
            Thread.sleep(60000); // keep container running for 60 seconds
        } catch (Exception e) {}
    }
}


Dockerfile

FROM eclipse-temurin:17-jdk

WORKDIR /app

COPY eg1.java ./

RUN javac eg1.java
 
CMD ["java","eg1"]
