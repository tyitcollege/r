using MongoDB.Bson;
using MongoDB.Bson.Serialization.Attributes;

public class UserModel
{
    [BsonId]
    [BsonRepresentation(BsonType.ObjectId)]
    public string Id { get; set; }

    public string Name { get; set; }

    public string Course { get; set; }
}

using MongoDB.Driver;
using System.Collections.Generic;

public class MongoDBHelper
{
    IMongoDatabase db;

    public MongoDBHelper()
    {
        var client = new MongoClient("mongodb+srv://arshiyafaisal:leekimv19@cluster0.prltjwz.mongodb.net/?appName=Cluster0");
        db = client.GetDatabase("WebFormsDB_Prac7");
    }

    public IMongoCollection<UserModel> Users
    {
        get { return db.GetCollection<UserModel>("UsersCollection"); }
    }

    public void InsertUser(UserModel user)
    {
        Users.InsertOne(user);
    }

    public List<UserModel> GetAllUsers()
    {
        return Users.Find(u => true).ToList();
    }

    public void UpdateUser(string id, UserModel user)
    {
        Users.ReplaceOne(u => u.Id == id, user);
    }

    public void DeleteUser(string id)
    {
        Users.DeleteOne(u => u.Id == id);
    }
}

using System;
using System.Web.UI;
using System.Web.UI.WebControls;

namespace pract7mongo2
{
    public partial class WebFormCRUD : System.Web.UI.Page
    {
        MongoDBHelper db = new MongoDBHelper();

        protected void Page_Load(object sender, EventArgs e)
        {
            if (!IsPostBack)
            {
                LoadUsers();
            }
        }

        void LoadUsers()
        {
            GridView1.DataSource = db.GetAllUsers();
            GridView1.DataBind();
        }

        protected void btnAdd_Click(object sender, EventArgs e)
        {
            UserModel user = new UserModel
            {
                Name = txtName.Text,
                Course = txtCourse.Text
            };

            db.InsertUser(user);

            txtName.Text = "";
            txtCourse.Text = "";

            LoadUsers();
        }

        protected void GridView1_RowEditing(object sender, GridViewEditEventArgs e)
        {
            GridView1.EditIndex = e.NewEditIndex;
            LoadUsers();
        }

        protected void GridView1_RowCancelingEdit(object sender, GridViewCancelEditEventArgs e)
        {
            GridView1.EditIndex = -1;
            LoadUsers();
        }

        protected void GridView1_RowUpdating(object sender, GridViewUpdateEventArgs e)
        {
            string id = GridView1.DataKeys[e.RowIndex].Value.ToString();

            GridViewRow row = GridView1.Rows[e.RowIndex];

            string name = ((TextBox)row.Cells[0].Controls[0]).Text;
            string course = ((TextBox)row.Cells[1].Controls[0]).Text;

            UserModel user = new UserModel
            {
                Id = id,
                Name = name,
                Course = course
            };

            db.UpdateUser(id, user);

            GridView1.EditIndex = -1;

            LoadUsers();
        }

        protected void GridView1_RowDeleting(object sender, GridViewDeleteEventArgs e)
        {
            string id = GridView1.DataKeys[e.RowIndex].Value.ToString();

            db.DeleteUser(id);

            LoadUsers();
        }
    }
}
